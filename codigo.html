/*
  Sensor Ultrasonico HC-SR04 con Arduino
  Extras a anadir: LiquidCrystal, TFT, Motores...
*/
int distancia = 0;
int max_distancia = 0;

long readUltrasonicDistance(int triggerPin, int echoPin)
{
  pinMode(triggerPin, OUTPUT);
  digitalWrite(triggerPin, LOW);
  delayMicroseconds(2);
  digitalWrite(triggerPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(triggerPin, LOW);
  pinMode(echoPin, INPUT);
  return pulseIn(echoPin, HIGH);
}

void setup()
{
  Serial.begin(9600);
  pinMode(5, OUTPUT);
  pinMode(7, OUTPUT);
  pinMode(7, OUTPUT);
  max_distancia = 300;
  distancia = 0;
}

void loop()
{
  distancia = 0.01723 * readUltrasonicDistance(13, 12);
  Serial.print("Distancia: ");
  Serial.print(distancia);
  Serial.print(" cm   y con ");
  Serial.print(max_distancia);
  Serial.println("cm de distancia maxima");

  if (distancia > max_distancia) {
    digitalWrite(5, LOW);
    digitalWrite(7, LOW);
  }
  if (distancia <= 15 && distancia <= max_distancia) {
    digitalWrite(5, HIGH);
    tone(7, 523, 1000);
  }
  if (distancia > 15) {
    digitalWrite(5, LOW);
    digitalWrite(7, LOW);
  }
  delay(2);
}
